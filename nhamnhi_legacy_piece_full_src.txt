    local dist = HubConfig.Farm.Distance or 6
    
    if mode == "Behind" then
        return targetPivot * CFrame.new(0, 0, dist)
    elseif mode == "Above" then
        return targetPivot * CFrame.new(0, dist, 0) * CFrame.Angles(math.rad(-90), 0, 0)

    end
    return targetPivot * CFrame.new(0, 0, dist)
end
